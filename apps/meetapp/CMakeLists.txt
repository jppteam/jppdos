# ---------------------------------------------------------------------------
# MeetApp — native binary app for the J++Device.
#
# The ESP-IDF component registration below is kept for IDE support (include
# paths, code navigation, static analysis) but this component is NOT linked
# into the firmware — nothing in main/ or jpp_core requires it. The firmware
# loads MeetApp at runtime from the SD card using jpp_native_loader_core.
#
# To build the loadable binary, run (from the project root, inside the Docker
# build environment):
#
#   idf.py build            # also triggers the meetapp_bin custom target
#
# The output is build/apps/meetapp/meetapp.bin (ELF32 shared object) and
# build/apps/meetapp/manifest.json — copy both to /sd/apps/meetapp/ on the SD card.
# ---------------------------------------------------------------------------

idf_component_register(
    SRCS
        "src/meetapp.c"
        "src/meetapp_entry.c"
        "src/meetapp_identity.c"
        "src/meetapp_ble.c"
        "src/meetapp_proof.c"
        "../common/jpp_ble_msg.c"
    INCLUDE_DIRS "include" "../common"
    REQUIRES jpp_core jpp_crypto_core freertos
)

# ---------------------------------------------------------------------------
# Standalone shared-library target (meetapp.bin).
#
# Uses the same riscv32-esp-elf toolchain that ESP-IDF picked up, with the
# -fPIC -shared -mno-relax flags required by the runtime ELF loader.
# Include paths are gathered from the ESP-IDF generator expressions so the
# build stays consistent with the firmware headers.
# ---------------------------------------------------------------------------

set(_MEETAPP_BIN ${CMAKE_BINARY_DIR}/apps/meetapp/meetapp.bin)

# build_shared.py reads compile_commands.json (generated by the main firmware
# build) to extract the exact -I/-D flags used for jpp_core, then recompiles
# MeetApp as a PIC shared library with the same include environment. This is
# more reliable than manually listing include paths across IDF versions.
#
# The target runs AFTER gen_project_binary so compile_commands.json is fully
# populated with the real compilation flags before the script reads it.
idf_build_get_property(_meetapp_python PYTHON)

add_custom_target(meetapp_bin ALL
    COMMAND ${_meetapp_python}
            ${CMAKE_CURRENT_SOURCE_DIR}/build_shared.py
            ${CMAKE_BINARY_DIR}
            ${CMAKE_CURRENT_SOURCE_DIR}
    COMMENT "Building MeetApp shared library → ${_MEETAPP_BIN}"
    DEPENDS gen_project_binary
    VERBATIM
)
